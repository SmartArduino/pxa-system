cmake_minimum_required(VERSION 3.20)
project(pxa_wamr_aux_stack_test C)
enable_testing()
get_filename_component(system_root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(WAMR_ROOT_DIR "${system_root}/wamr")
set(PXA_WASI_SDK_DIR "" CACHE PATH "WASI SDK containing bin/clang")
set(PXA_WAMRC "" CACHE FILEPATH "Matching wamrc for the optional AOT cases")
option(PXA_TEST_PARTITIONED_STACK "Historical control; bypass production policy" OFF)
if(NOT EXISTS "${PXA_WASI_SDK_DIR}/bin/clang")
    message(FATAL_ERROR "Set PXA_WASI_SDK_DIR to a WASI SDK")
endif()
set(WAMR_BUILD_PLATFORM linux)
set(WAMR_BUILD_TARGET X86_64)
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 1)
set(WAMR_BUILD_LIBC_BUILTIN 1)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_LIB_PTHREAD 1)
set(WAMR_BUILD_LIB_WASI_THREADS 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 1)
set(WAMR_BUILD_SHARED_MEMORY 1)
set(WAMR_BUILD_DUMP_CALL_STACK 0)
set(WAMR_BUILD_BULK_MEMORY 1)
include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")
add_library(wamr STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_link_libraries(wamr PUBLIC pthread m dl)
if(PXA_TEST_PARTITIONED_STACK)
    include("${system_root}/cmake/PxaWamrLoader.cmake")
    pxa_configure_wamr_loader(wamr "${WAMR_ROOT_DIR}")
    target_compile_definitions(wamr PRIVATE WASM_ENABLE_HEAP_AUX_STACK_ALLOCATION=0)
    set(expect_full 0)
else()
    include("${system_root}/cmake/PxaWamrRuntime.cmake")
    pxa_configure_wamr_runtime(wamr "${WAMR_ROOT_DIR}")
    set(expect_full 1)
endif()
add_executable(check check.c)
target_link_libraries(check PRIVATE wamr)
target_compile_definitions(check PRIVATE PXA_TEST_FULL_AUX_STACK=${expect_full})
target_compile_options(check PRIVATE -UNDEBUG -Wall -Wextra -Werror)
foreach(name guest thread_guest)
    set(wasm "${CMAKE_CURRENT_BINARY_DIR}/${name}.wasm")
    set(shared_flags)
    if(name STREQUAL thread_guest)
        list(APPEND shared_flags -matomics -mbulk-memory -Wl,--shared-memory)
    endif()
    add_custom_command(OUTPUT "${wasm}"
        COMMAND "${PXA_WASI_SDK_DIR}/bin/clang" --target=wasm32 -O2 -nostdlib
            ${shared_flags} -Wl,--no-entry -Wl,--stack-first
            -Wl,-z,stack-size=16384 -Wl,--export=__heap_base
            -Wl,--export=__data_end -Wl,--initial-memory=65536
            -Wl,--max-memory=131072 "${CMAKE_CURRENT_LIST_DIR}/${name}.c" -o "${wasm}"
        DEPENDS "${name}.c" guest.c VERBATIM)
    add_custom_target(${name}_wasm ALL DEPENDS "${wasm}")
    add_test(NAME ${name}_wasm COMMAND check "${wasm}")
    if(PXA_WAMRC AND EXISTS "${PXA_WAMRC}")
        set(aot "${CMAKE_CURRENT_BINARY_DIR}/${name}.aot")
        add_custom_command(OUTPUT "${aot}"
            COMMAND "${PXA_WAMRC}" --target=x86_64 --bounds-checks=1
                --opt-level=3 --size-level=0 -o "${aot}" "${wasm}"
            DEPENDS "${wasm}" VERBATIM)
        add_custom_target(${name}_aot ALL DEPENDS "${aot}")
        add_test(NAME ${name}_aot COMMAND check "${aot}")
    endif()
endforeach()
