if(NOT COMMAND idf_component_register)
    cmake_minimum_required(VERSION 3.16)
endif()

include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/PxaSystemSources.cmake")

if(COMMAND idf_component_register)
    idf_component_register(
        SRCS ${PXSYS_STANDARD_SOURCES}
        INCLUDE_DIRS "${CMAKE_CURRENT_LIST_DIR}/include"
        REQUIRES pxa_system_core pxa_system_native)
else()
    if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        project(pxa_system_standard VERSION 0.1.0 LANGUAGES C)
    endif()
    if(NOT TARGET pxa_system_core)
        add_subdirectory("${PXSYS_ROOT}/system/core" "pxa_system_core")
    endif()
    if(NOT TARGET pxa_system_native)
        add_subdirectory("${PXSYS_ROOT}/system/runtimes/native" "pxa_system_native")
    endif()
    add_library(pxa_system_standard ${PXSYS_STANDARD_SOURCES})
    add_library(pxsys::standard ALIAS pxa_system_standard)
    set_target_properties(pxa_system_standard PROPERTIES EXPORT_NAME standard)
    target_compile_features(pxa_system_standard PUBLIC c_std_99)
    target_include_directories(pxa_system_standard PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>)
    target_link_libraries(pxa_system_standard PUBLIC pxsys::core pxsys::native)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(pxa_system_standard PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(PXSYS_BUILD_TESTS)
    add_executable(pxsys_standard_system_test tests/standard_system_test.c)
    target_link_libraries(pxsys_standard_system_test PRIVATE pxsys::standard)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(pxsys_standard_system_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME pxsys_standard_system_test COMMAND pxsys_standard_system_test)
    endif()
endif()
