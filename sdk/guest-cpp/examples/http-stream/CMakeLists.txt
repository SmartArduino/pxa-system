cmake_minimum_required(VERSION 3.20)
project(pxa_http_stream CXX)
set(PXA_LINEAR_MEMORY_MAXIMUM 131072)
list(APPEND CMAKE_MODULE_PATH "${PXA_CMAKE_MODULE_DIR}")
include(PxaGuest)
pxa_add_app(pxa_main COMPONENT_ID main SOURCES main.cpp)
set(_fixture_default "http://127.0.0.1:31877")
if(DEFINED ENV{PXA_HTTP_FIXTURE_URL})
    set(PXA_HTTP_FIXTURE_URL "$ENV{PXA_HTTP_FIXTURE_URL}" CACHE STRING "HTTP validation fixture URL" FORCE)
endif()
set(PXA_HTTP_FIXTURE_URL "${_fixture_default}" CACHE STRING "HTTP validation fixture URL")
target_compile_definitions(pxa_main PRIVATE PXA_HTTP_FIXTURE_URL="${PXA_HTTP_FIXTURE_URL}")
if(DEFINED ENV{PXA_HTTP_TEST_CASE})
    set(PXA_HTTP_TEST_CASE "$ENV{PXA_HTTP_TEST_CASE}" CACHE STRING "Empty: full validation; buffered or streamed: matched 256 KiB baseline" FORCE)
else()
    set(PXA_HTTP_TEST_CASE "" CACHE STRING "Empty: full validation; buffered or streamed: matched 256 KiB baseline")
endif()
if(PXA_HTTP_TEST_CASE STREQUAL "buffered")
    target_compile_definitions(pxa_main PRIVATE PXA_HTTP_BASELINE=1)
elseif(PXA_HTTP_TEST_CASE STREQUAL "streamed")
    target_compile_definitions(pxa_main PRIVATE PXA_HTTP_BASELINE=2)
elseif(PXA_HTTP_TEST_CASE STREQUAL "paired")
    target_compile_definitions(pxa_main PRIVATE PXA_HTTP_BASELINE=3)
else()
    target_compile_definitions(pxa_main PRIVATE PXA_HTTP_BASELINE=0)
endif()
