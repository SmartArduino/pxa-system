cmake_minimum_required(VERSION 3.20)
project(pxa_local_network CXX)
set(PXA_LINEAR_MEMORY_MAXIMUM 131072)
list(APPEND CMAKE_MODULE_PATH "${PXA_CMAKE_MODULE_DIR}")
include(PxaGuest)
pxa_add_app(pxa_main COMPONENT_ID main SOURCES main.cpp)
set(PXA_LAN_DEMO_ROLE "server" CACHE STRING "server or client")
if(DEFINED ENV{PXA_LAN_DEMO_ROLE})
    set(PXA_LAN_DEMO_ROLE "$ENV{PXA_LAN_DEMO_ROLE}" CACHE STRING "server or client" FORCE)
endif()
if(PXA_LAN_DEMO_ROLE STREQUAL "server")
    target_compile_definitions(pxa_main PRIVATE PXA_LAN_DEMO_SERVER=1)
elseif(PXA_LAN_DEMO_ROLE STREQUAL "client")
    target_compile_definitions(pxa_main PRIVATE PXA_LAN_DEMO_SERVER=0)
else()
    message(FATAL_ERROR "PXA_LAN_DEMO_ROLE must be server or client")
endif()
set(PXA_LAN_DEMO_ENDPOINT "" CACHE STRING "Optional direct TCP IPv4 URL; empty uses discovery")
set(PXA_LAN_DEMO_SCOPE "local-network" CACHE STRING "Listener/datagram permission scope")
foreach(option ENDPOINT SCOPE)
    if(DEFINED ENV{PXA_LAN_DEMO_${option}})
        set(PXA_LAN_DEMO_${option} "$ENV{PXA_LAN_DEMO_${option}}" CACHE STRING "Network demo option" FORCE)
    endif()
endforeach()
if(NOT PXA_LAN_DEMO_SCOPE STREQUAL "local-network" AND NOT PXA_LAN_DEMO_SCOPE STREQUAL "network")
    message(FATAL_ERROR "Scope must be local-network or network")
endif()
target_compile_definitions(pxa_main PRIVATE
    PXA_LAN_DEMO_ENDPOINT="${PXA_LAN_DEMO_ENDPOINT}"
    PXA_LAN_DEMO_SCOPE="${PXA_LAN_DEMO_SCOPE}"
    PXA_LAN_DEMO_PORT=31888)
