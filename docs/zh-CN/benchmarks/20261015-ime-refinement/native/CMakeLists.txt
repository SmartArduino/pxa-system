cmake_minimum_required(VERSION 3.20)
project(ime_synchronous_memory C)
enable_testing()
get_filename_component(SDK "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
get_filename_component(PRODUCT "${SDK}/../.." ABSOLUTE)
set(LVGL_SOURCE_DIR "${PRODUCT}/firmware/managed_components/lvgl__lvgl" CACHE PATH
    "Existing product LVGL source tree")
find_package(Freetype REQUIRED)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(LV_BUILD_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
target_include_directories(lvgl PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${LVGL_SOURCE_DIR}")
target_link_libraries(lvgl PUBLIC Freetype::Freetype m)
set(PXSYS_BUILD_LIBPXA OFF CACHE BOOL "" FORCE)
set(PXSYS_BUILD_WAMR_ADAPTER OFF CACHE BOOL "" FORCE)
set(PXSYS_BUILD_PXA_RUNTIME OFF CACHE BOOL "" FORCE)
set(PXSYS_BUILD_SIMULATOR OFF CACHE BOOL "" FORCE)
set(PXSYS_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(PXSYS_BUILD_REFERENCE_LVGL ON CACHE BOOL "" FORCE)
set(PXSYS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDK}" pxa-system)
